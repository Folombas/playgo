### Физика корабля

```go
// Поворот
if KeyLeft { p.angle -= PlayerTurnSpeed }
if KeyRight { p.angle += PlayerTurnSpeed }

// Тяга
if KeyUp {
    p.vx += math.Cos(p.angle) * 0.3
    p.vy += math.Sin(p.angle) * 0.3
}

// Трение
p.vx *= 0.98
p.vy *= 0.98
```

### Самонаводящиеся пули
```go
if bullet.homing {
    // Найти ближайшего врага
    for _, e := range enemies {
        dist := math.Hypot(e.x-b.x, e.y-b.y)
        if dist < minDist {
            target = e
        }
    }
    // Повернуть к цели
    if target != nil {
        angle := math.Atan2(target.y-b.y, target.x-b.x)
        b.angle += (angle - b.angle) * 0.1
    }
}
```
