## 🔧 Технические детали

### Система карт
```go
type Card struct {
    id          int
    name        string
    description string
    cardType    CardType  // Attack, Skill, Power
    rarity      CardRarity // Common, Uncommon, Rare
    cost        int
    damage      int
    block       int
}
```

### Боевой цикл
```go
func (g *Game) playCard(card *Card, target *Enemy) {
    if p.energy < card.cost {
        return
    }
    
    p.energy -= card.cost
    
    switch card.cardType {
    case CardAttack:
        target.health -= card.damage
    case CardSkill:
        p.block += card.block
    case CardPower:
        // Permanent buff
    }
    
    // Move to discard
}
```

### ИИ врагов
```go
// Простой ИИ - всегда атака
enemy.intent = "attack"
enemy.intentValue = enemy.damage

// Нанесение урона
damage := enemy.damage
block := g.player.block
if block >= damage {
    g.player.block -= damage
} else {
    g.player.health -= (damage - block)
    g.player.block = 0
}
```
